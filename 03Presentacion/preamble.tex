\newcommand{\cajaarchivofaltante}[1]{%
  \begin{center}
  \fcolorbox{red!60!black}{red!5}{\ttfamily\scriptsize ARCHIVO FALTANTE: \detokenize{#1}}
  \end{center}}

\newcommand{\cajaimagenfaltante}[1]{%
  \begingroup\setlength{\fboxsep}{0pt}%
  \fcolorbox{tecrule}{boxbg}{%
    \parbox[c][1.2cm][c]{2.6cm}{\centering\ttfamily\tiny
      IMAGEN NO ENCONTRADA\\[2pt]\detokenize{#1}}}%
  \endgroup}

\newcommand{\safeinput}[1]{%
  \IfFileExists{#1.tex}{\input{#1}}{%
    \PackageWarning{tecslides}{Falta el archivo #1.tex}%
    \begin{frame}\cajaarchivofaltante{#1.tex}\end{frame}}}

\newcommand{\safeimage}[2][height=1.2cm]{%
  \IfFileExists{#2}{\includegraphics[#1]{#2}}{%
    \IfFileExists{img/#2}{\includegraphics[#1]{img/#2}}{%
      \PackageWarning{tecslides}{Falta la imagen #2}%
      \cajaimagenfaltante{#2}}}}
