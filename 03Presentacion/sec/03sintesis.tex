\begin{frame}{Eje 1: el error no es igual para todos}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \begin{block}{Auditor\'ia comercial \cite{buolamwini_gendershades_2018}}
        \begin{columns}
          \begin{column}{0.48\linewidth}
            \cifra{34.7\,\%}{error m\'aximo, mujeres de piel oscura}
          \end{column}
          \begin{column}{0.48\linewidth}
            \cifra{0.8\,\%}{hombres de piel clara}
          \end{column}
        \end{columns}
        \scriptsize Tras publicarse, los proveedores se\~nalados redujeron
        sus brechas \cite{raji_actionable_2019}.
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Evaluaci\'on NIST \cite{grother_nistir8280_2019}}
        \cifra{10 a $>$100$\times$}{m\'as falsos positivos seg\'un el grupo}
        \scriptsize 189 algoritmos de 99 desarrolladores; la magnitud var\'ia
        mucho entre algoritmos.
      \end{block}
    \end{column}
  \end{columns}

  \begin{alertblock}{Lectura cruzada}
    \scriptsize
    Coinciden en la direcci\'on, no en la magnitud: uno mide
    clasificaci\'on de sexo sobre 1\,270 retratos; el otro, identificaci\'on
    sobre 18.27 millones de im\'agenes. Las cifras no se comparan
    directamente.
  \end{alertblock}
\end{frame}

\begin{frame}{Eje 2: la integridad del sistema es atacable}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Ataque f\'isico \cite{sharif_accessorize_2016}}
        \scriptsize
        Anteojos con patrones impresos permiten evadir la identificaci\'on o
        suplantar a otra persona.\\[4pt]
        \textbf{Amenaza:} integridad.\\
        \textbf{Control:} detecci\'on de ataques de presentaci\'on
        (ISO/IEC 30107-3).
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Protecci\'on del usuario \cite{shan_fawkes_2020}}
        \scriptsize
        Perturbaciones imperceptibles en fotos propias antes de publicarlas
        impiden entrenar modelos no autorizados.\\[4pt]
        \textbf{Protege:} confidencialidad.\\
        \textbf{L\'imite:} supone un atacante que no se adapta.
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.2cm}
  \begin{center}
    \scriptsize\color{tecnavy}\textbf{Ataque y defensa se juegan en el mismo
    espacio: la representaci\'on del rostro.}
  \end{center}
\end{frame}

\begin{frame}{Eje 3: del laboratorio a la calle}
  \begin{columns}[T]

    \begin{column}{0.46\textwidth}
      \begin{block}{Despliegue real \cite{fussey_essex_2019}}

        pruebas de Londres \pct}
      \end{block}
    \end{column}
  \end{columns}

  \begin{alertblock}{Lectura cruzada}
    \scriptsize
    La exactitud medida con im\'agenes controladas no se traslada al
    despliegue en espacio p\'ublico: cambian la iluminaci\'on, la pose, la
    base de referencia y la tasa base de personas buscadas.
  \end{alertblock}
\end{frame}

\begin{frame}{Contradicciones, vac\'ios y conexi\'on con el curso}
